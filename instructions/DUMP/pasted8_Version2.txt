function uploadImage($filePath, $galleryId, $db) {
    global $r2;
    $filename = basename($filePath);

    // Upload original
    $r2->putObject([
        'Bucket' => 'tiredofdointm',
        'Key' => "gallery_$galleryId/original/$filename",
        'SourceFile' => $filePath,
        'ACL' => 'private',
    ]);

    // Create derivatives
    $thumbPath = createThumbnail($filePath);
    $webPath = createWebOptimized($filePath);
    $mobilePath = createMobileOptimized($filePath);

    $r2->putObject([
        'Bucket' => 'tiredofdointm',
        'Key' => "gallery_$galleryId/thumb/$filename",
        'SourceFile' => $thumbPath,
        'ACL' => 'public-read',
    ]);

    $r2->putObject([
        'Bucket' => 'tiredofdointm',
        'Key' => "gallery_$galleryId/web/$filename",
        'SourceFile' => $webPath,
        'ACL' => 'public-read',
    ]);

    $r2->putObject([
        'Bucket' => 'tiredofdointm',
        'Key' => "gallery_$galleryId/mobile/$filename",
        'SourceFile' => $mobilePath,
        'ACL' => 'public-read',
    ]);

    // Save in DB
    $stmt = $db->prepare("INSERT INTO images (gallery_id, original_r2_url, thumbnail_url, web_url, mobile_url, processed, uploaded_at) VALUES (?, ?, ?, ?, ?, 1, NOW())");
    $stmt->execute([
        $galleryId,
        "gallery_$galleryId/original/$filename",
        "gallery_$galleryId/thumb/$filename",
        "gallery_$galleryId/web/$filename",
        "gallery_$galleryId/mobile/$filename"
    ]);
}

function createThumbnail($file) {
    $thumb = 'thumb_' . basename($file);
    exec("convert $file -resize 300x300 $thumb");
    return $thumb;
}

function createWebOptimized($file) {
    $web = 'web_' . basename($file);
    exec("convert $file -resize 1200x1200 $web");
    return $web;
}

function createMobileOptimized($file) {
    $mobile = 'mobile_' . basename($file);
    exec("convert $file -resize 800x800 $mobile");
    return $mobile;
}

  $ciphertext = openssl_encrypt($plaintext, 'aes-256-gcm', $masterKey, OPENSSL_RAW_DATA, $iv, $tag);

function decryptField($blob, $masterKey) {

  $iv = substr($data, 0, 12);
  $tag = substr($data, 12, 16);

  return openssl_decrypt($ciphertext, 'aes-256-gcm', $masterKey, OPENSSL_RAW_DATA, $iv, $tag);
}

2️⃣ DATABASE SCHEMAS (PostgreSQL)
-- USERS
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    role_id INT NOT NULL REFERENCES roles(id),
    avatar_url TEXT,
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- ROLES
CREATE TABLE roles (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50) UNIQUE NOT NULL -- Admin, Staff, Manager, Client
);

-- SESSIONS
CREATE TABLE sessions (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    token TEXT UNIQUE NOT NULL,
    expires_at TIMESTAMP NOT NULL,
    created_at TIMESTAMP DEFAULT now()
);

-- GALLERIES
CREATE TABLE galleries (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    type VARCHAR(20) NOT NULL, -- public / link / private
    cover_image_id INT REFERENCES images(id),
    created_by INT REFERENCES users(id),
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- GALLERY LINKS (for link-only galleries)
CREATE TABLE gallery_links (
    id SERIAL PRIMARY KEY,
    gallery_id INT REFERENCES galleries(id) ON DELETE CASCADE,
    token VARCHAR(255) UNIQUE NOT NULL,
    permissions JSONB DEFAULT '{}'::jsonb, -- view/edit
    created_at TIMESTAMP DEFAULT now(),
    expires_at TIMESTAMP
);

-- IMAGES
CREATE TABLE images (
    id SERIAL PRIMARY KEY,
    gallery_id INT REFERENCES galleries(id) ON DELETE CASCADE,
    uploader_id INT REFERENCES users(id),
    filename_raw TEXT NOT NULL,
    filename_edited TEXT NOT NULL,
    caption TEXT,
    location JSONB, -- {lat:.., lng:.., place:..}
    metadata JSONB,
    created_at TIMESTAMP DEFAULT now()
);

-- BOOKINGS
CREATE TABLE bookings (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id),
    service_type VARCHAR(100),
    date_start TIMESTAMP NOT NULL,
    date_end TIMESTAMP NOT NULL,
    status VARCHAR(20) DEFAULT 'requested', -- requested / confirmed / paid / completed / cancelled
    google_event_id TEXT,
    created_at TIMESTAMP DEFAULT now()
);

-- PAYMENTS
CREATE TABLE payments (
    id SERIAL PRIMARY KEY,
    booking_id INT REFERENCES bookings(id),
    method VARCHAR(20), -- stripe / paypal
    amount NUMERIC(10,2),
    status VARCHAR(20), -- pending / completed / failed
    transaction_id TEXT,
    created_at TIMESTAMP DEFAULT now()
);

-- ANALYTICS EVENTS
CREATE TABLE analytics_events (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id),
    event_type VARCHAR(100),
    details JSONB,
    created_at TIMESTAMP DEFAULT now()
);

-- ALERTS
CREATE TABLE alerts (
    id SERIAL PRIMARY KEY,
    type VARCHAR(50),
    message TEXT,
    metadata JSONB,
    read BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT now()
);
