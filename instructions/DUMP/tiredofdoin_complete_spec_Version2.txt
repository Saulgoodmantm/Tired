SQL
CREATE TABLE users (
  id SERIAL PRIMARY KEY,
  username VARCHAR(50) UNIQUE,
  email VARCHAR(255) UNIQUE NOT NULL,
  role_id INT REFERENCES roles(id),
  avatar_url TEXT,
  legal_names JSONB, -- Store multiple, never delete old
  emails JSONB, -- Store multiple
  ips JSONB, -- Store multiple for security
  device_fingerprints JSONB,
  stripe_customer_id VARCHAR(255),
  has_booked BOOLEAN DEFAULT FALSE,
  first_booking_date TIMESTAMP,
  previous_galleries JSONB,
  created_at TIMESTAMP DEFAULT now(),
  updated_at TIMESTAMP DEFAULT now(),
  verified BOOLEAN DEFAULT FALSE,
  last_login TIMESTAMP
);
roles

SQL
CREATE TABLE roles (
  id SERIAL PRIMARY KEY,
  name VARCHAR(50) UNIQUE NOT NULL,
  permissions JSONB NOT NULL
);
sessions

SQL
CREATE TABLE sessions (
  id SERIAL PRIMARY KEY,
  user_id INT REFERENCES users(id),
  token TEXT UNIQUE NOT NULL,
  device_fingerprint TEXT,
  ip_address VARCHAR(50),
  user_agent TEXT,
  expires_at TIMESTAMP NOT NULL,
  created_at TIMESTAMP DEFAULT now()

SQL
CREATE TABLE galleries (
  id SERIAL PRIMARY KEY,

  type VARCHAR(20) CHECK (type IN ('public', 'unlisted', 'private')),

  cover_image_id INT REFERENCES images(id),
  created_by INT REFERENCES users(id),
  shoot_date DATE,
  location_id INT REFERENCES locations(id),
  is_pinned BOOLEAN DEFAULT FALSE,
  pin_order INT,
  created_at TIMESTAMP DEFAULT now(),
  updated_at TIMESTAMP DEFAULT now()
);

Encryption Implementation
JavaScript
// Encryption wrapper
const crypto = require('crypto');

function encrypt(data, context) {
  const key = getKeyForContext(context); // Different keys per data type
  const iv = crypto. randomBytes(16);
  const cipher = crypto.createCipheriv('aes-256-gcm', key, iv);
  
  let encrypted = cipher.update(JSON.stringify(data), 'utf8', 'hex');
  encrypted += cipher.final('hex');
  
  const authTag = cipher.getAuthTag();
  
  return {
    encrypted,
    iv: iv.toString('hex'),
    authTag: authTag.toString('hex'),
    context // Store context for key retrieval during decryption
  };
}

  // 1. Verify admin role

Use JSONB for flexible metadata (image EXIF, user preferences)

This system uses passwordless authentication exclusively. Users never create or manage passwords.

Why Passwordless?
No password breaches — Can't leak what you don't store
Email verification built-in — Every login verifies email ownership
Simpler UX — No password reset flows
Appropriate for this use case — Photography site, not a bank

function generateOTP() {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; // Removed ambiguous:  0,O,I,1,L
  let otp = '';
  for (let i = 0; i < 6; i++) {
    otp += chars.charAt(crypto.randomInt(chars.length));
  }
  return otp;
}

Session Management
Session Creation
JavaScript
async function createSession(user, request) {
  // 1. Generate cryptographically secure token
  const sessionToken = crypto.randomBytes(32).toString('hex');
  
  // 2. Generate device fingerprint
  const fingerprint = generateFingerprint(request);
  
  // 3. Calculate expiration (30 days)
  const expiresAt = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000);
  
  // 4. Store session
  const session = await db.sessions.create({
    user_id: user.id,
    token: hashToken(sessionToken), // Store hashed, not plain
    device_fingerprint: fingerprint,
    ip_address: request.ip,
    user_agent: request.headers['user-agent'],
    expires_at: expiresAt,
    created_at: new Date()
  });
  
  // 5. Set secure cookie
  response.cookie('session', sessionToken, {
    httpOnly: true,        // Not accessible via JavaScript
    secure: true,          // HTTPS only
    sameSite: 'lax',       // CSRF protection
    maxAge: 30 * 24 * 60 * 60 * 1000, // 30 days
    domain: 'tiredofdointm. com'
  });
  
  // 6. Regenerate session ID (prevent fixation)
  request.session.regenerate();
  
  return session;
}

function generateFingerprint(request) {
  const components = [
    request.headers['user-agent'],
    request.headers['accept-language'],
    request.headers['accept-encoding'],
    // Add more stable browser characteristics
  ];
  
  return crypto
    .createHash('sha256')
    .update(components.join('|'))
    .digest('hex')
    .substring(0, 32);
}
Session Validation (Every Request)
JavaScript
async function validateSession(request) {
  // 1. Get token from cookie
  const token = request.cookies.session;
  if (!token) return null;
  
  // 2. Find session by hashed token
  const session = await db.sessions.findOne({
    token: hashToken(token)
  });
  
  if (!session) return null;
  
  // 3. Check expiration
  if (new Date() > session.expires_at) {
    await db.sessions. delete(session. id);
    return null;
  }
  
  // 4. Validate fingerprint (optional security)
  const currentFingerprint = generateFingerprint(request);
  if (session.device_fingerprint !== currentFingerprint) {
    // Fingerprint mismatch - possible session theft
    await logSecurityEvent('fingerprint_mismatch', session. user_id, {
      expected: session.device_fingerprint,
      actual: currentFingerprint,
      ip: request.ip
    });
    
    // Option A:  Invalidate session (more secure)
    // await db.sessions.delete(session.id);
    // return null;
    
    // Option B: Allow but flag (less disruptive)
    session.flagged = true;
  }
  
  // 5. Get user
  const user = await db.users.findById(session.user_id);
  if (!user) return null;
  
  // 6. Attach to request
  request.user = user;
  request.session = session;
  
  return user;
}

Security Implementation
Why HMAC Cookie?
A simple plaintext cookie could be modified. HMAC ensures integrity.

JavaScript
const GATE_SECRET = process.env.GATE_SECRET; // Random 32+ char string

function createGateCookie() {
  const timestamp = Date.now();
  const expiresAt = timestamp + (30 * 24 * 60 * 60 * 1000); // 30 days
  
  const payload = `${timestamp}:${expiresAt}`;
  const signature = crypto
    .createHmac('sha256', GATE_SECRET)
    .update(payload)
    .digest('hex');
  
  return `${payload}:${signature}`;
}

function validateGateCookie(cookie) {
  if (!cookie) return false;
  
  const parts = cookie.split(':');
  if (parts.length !== 3) return false;
  
  const [timestamp, expiresAt, signature] = parts;
  
  // Check expiration
  if (Date.now() > parseInt(expiresAt)) return false;
  
  // Verify signature
  const expectedSignature = crypto
    .createHmac('sha256', GATE_SECRET)
    .update(`${timestamp}:${expiresAt}`)
    .digest('hex');
  
  return crypto.timingSafeEqual(
    Buffer.from(signature),
    Buffer.from(expectedSignature)
  );
}

// The actual validation happens server-side
app.post('/api/gate/validate', (req, res) => {
  const { password } = req.body;
  
  // Constant-time comparison to prevent timing attacks
  const expected = Buffer.from('67');
  const received = Buffer.from(password || '');
  
  if (expected.length !== received.length) {
    return res.status(401).json({ error: 'Invalid password' });
  }
  
  if (!crypto.timingSafeEqual(expected, received)) {
    return res.status(401).json({ error: 'Invalid password' });
  }
  
  // Set HMAC cookie
  res.cookie('gate_pass', createGateCookie(), {
    httpOnly: true,
    secure: true,
    sameSite: 'lax',
    maxAge: 30 * 24 * 60 * 60 * 1000
  });
  
  res.json({ success: true });
});

Permission Check Middleware
JavaScript
function requirePermission(permission) {
  return async (req, res, next) => {
    const user = req. user;
    
    // Guest handling
    if (!user) {
      if (permission === 'view_public') {
        return next();
      }
      return res.redirect(`/auth? next=${encodeURIComponent(req.originalUrl)}`);
    }
    
    // Get role permissions
    const role = await db.roles.findById(user.role_id);
    
    // Admin wildcard
    if (role. permissions['*']) {
      return next();
    }
    
    // Check specific permission
    if (role.permissions[permission]) {
      return next();
    }
    
    // Permission denied
    logSecurityEvent('permission_denied', user.id, {
      permission,
      route: req.originalUrl
    });
    
    return res.status(403).render('errors/403', {
      message: 'You do not have permission to access this page'
    });
  };
}

🎞️ HERO SLIDESHOW SYSTEM
Overview
The homepage hero showcases pinned photography with a cinematic, GTA V loading screen-style animation. Images slowly pan and zoom, creating constant motion.

Animation Specification
State Machine
Code
┌─────────────────┐
│   ENTERING      │ ← Image slides in quickly
│   (300ms)       │
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│   DISPLAYING    │ ← Slow pan + zoom (4-7 seconds)
│   (4000-7000ms) │
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│   EXITING       │ ← Speed ramps up, slides out
│   (300ms)       │
└────────┬────────┘
         │
         ▼
    [Next Image]
Animation Parameters (Per Image, Configurable)
TypeScript
interface SlideshowConfig {
  // Entry
  entryDirection: 'left' | 'right' | 'top' | 'bottom';
  entryDuration: number; // 200-400ms
  
  // Display
  displayDuration: number; // 4000-7000ms (randomized within range)
  
  // Pan
  panDirection: 'left' | 'right' | 'up' | 'down' | 'diagonal-*';
  panDistance: number; // Percentage of image width/height (5-15%)
  
  // Zoom
  zoomStart: number; // 1. 0-1.1
  zoomEnd: number; // 1.05-1.15
  
  // Exit
  exitDirection: 'left' | 'right' | 'top' | 'bottom';
  exitDuration: number; // 200-400ms
}
GSAP Implementation
JavaScript
class HeroSlideshow {
  constructor(container, images, configs) {
    this.container = container;
    this.images = images;
    this.configs = configs;
    this.currentIndex = 0;
    this. isPlaying = true;
    
    // Preload next 2 images
    this.preloadQueue = [];
    
    this.init();
  }
  
  init() {
    // Create image elements
    this.createImageElements();
    
    // Start slideshow
    this.showImage(0);
  }
  
  createImageElements() {
    this.imageElements = this.images.map((src, i) => {
      const wrapper = document.createElement('div');
      wrapper.className = 'slideshow-image-wrapper';
      wrapper.style.cssText = `
        position: absolute;
        inset: 0;
        overflow: hidden;
        opacity: 0;
      `;
      
      const img = document.createElement('img');
      img.src = src;
      img.className = 'slideshow-image';
      img.style.cssText = `
        width: 100%;
        height: 100%;
        object-fit: cover;
      `;
      
      wrapper.appendChild(img);
      this.container.appendChild(wrapper);
      
      return { wrapper, img };
    });
  }
  
  async showImage(index) {
    if (! this.isPlaying) return;
    
    const config = this.configs[index] || this.getDefaultConfig();
    const { wrapper, img } = this. imageElements[index];
    const previousWrapper = this.imageElements[this.currentIndex]?.wrapper;
    
    // Calculate random display duration
    const displayDuration = 4000 + Math.random() * 3000;
    
    // Timeline for this image
    const tl = gsap.timeline();
    
    // Entry animation
    const entryOffset = this.getEntryOffset(config.entryDirection);
    gsap.set(wrapper, { opacity: 1 });
    gsap.set(img, { 
      x: entryOffset.x, 
      y: entryOffset.y,
      scale: config.zoomStart 
    });
    
    tl.to(img, {
      x: 0,
      y: 0,
      duration: config.entryDuration / 1000,
      ease: 'power2.out'
    });
    
    // Pan and zoom during display
    const panOffset = this.getPanOffset(config.panDirection, config.panDistance);
    tl.to(img, {
      x: panOffset.x,
      y: panOffset.y,
      scale: config.zoomEnd,
      duration: displayDuration / 1000,
      ease: 'none'
    });
    
    // Exit animation (runs concurrently with next entry)
    const exitOffset = this.getExitOffset(config. exitDirection);
    tl.to(img, {
      x: exitOffset.x,
      y: exitOffset.y,
      duration: config.exitDuration / 1000,
      ease: 'power2.in'
    });
    tl.set(wrapper, { opacity: 0 });
    
    // Queue next image
    const nextIndex = (index + 1) % this.images.length;
    
    // Start next image slightly before current exits
    setTimeout(() => {
      this.currentIndex = nextIndex;
      this.showImage(nextIndex);
    }, config.entryDuration + displayDuration);
    
    // Preload upcoming images
    this.preloadImage((nextIndex + 1) % this.images.length);
    this.preloadImage((nextIndex + 2) % this.images.length);
  }
  
  getEntryOffset(direction) {
    const offsets = {
      'left': { x: '-100%', y: 0 },
      'right':  { x: '100%', y: 0 },
      'top': { x: 0, y: '-100%' },
      'bottom': { x: 0, y: '100%' }
    };
    return offsets[direction] || offsets.left;
  }
  
  getPanOffset(direction, distance) {
    const d = `${distance}%`;
    const offsets = {
      'left': { x: `-${d}`, y: 0 },
      'right': { x: d, y: 0 },
      'up': { x: 0, y: `-${d}` },
      'down': { x: 0, y: d },
      'diagonal-up-left': { x: `-${d}`, y: `-${d}` },
      'diagonal-up-right': { x: d, y: `-${d}` },
      'diagonal-down-left': { x: `-${d}`, y: d },
      'diagonal-down-right': { x: d, y:  d }
    };
    return offsets[direction] || offsets.right;
  }
  
  getDefaultConfig() {
    return {
      entryDirection: 'left',
      entryDuration:  300,
      panDirection: 'right',
      panDistance: 10,
      zoomStart: 1.05,
      zoomEnd: 1.12,
      exitDirection: 'left',
      exitDuration:  300
    };
  }
  
  preloadImage(index) {
    if (this.preloadQueue.includes(index)) return;
    
    const img = new Image();
    img.src = this.images[index];
    this.preloadQueue.push(index);
  }
  
  pause() {
    this.isPlaying = false;
  }
  
  play() {
    this.isPlaying = true;
    this.showImage(this.currentIndex);
  }
}

🔒 GLOBAL SYSTEM RULES {#global-system-rules}
These rules apply to EVERY component, page, API call, and interaction.

Rule 1: Permission-First Architecture
Every request must:

Identify the user (session/token/guest)
Determine their role
Check permission for requested resource
Allow OR deny with logging
Implementation Pattern:

Code
Request → Identify User → Check Role → Verify Permission → Proceed/Deny
Edge Cases:

Expired session during action → Re-authenticate, retry
Role changed while logged in → Immediate permission refresh
Shared link access → Validate link token separately
API key access → Different permission path
Rule 2: Auth Gate on All Protected Routes
Every navigation to protected pages must route through /auth:

Code
User clicks /dashboard
    ↓
Redirect to /auth? next=/dashboard
    ↓
Validate session + role
    ↓
Success → Redirect to /dashboard
Failure → Show login with next= preserved
Why This Matters:

Prevents bookmark bypass
Prevents shared link abuse
Graceful session timeout handling
Consistent authentication flow
Rule 3: Data Classification
Public Data (Never Encrypted):

Data	Cache Duration
Public gallery thumbnails	7 days
Pricing information	1 hour
Testimonials	1 day
Homepage content	1 day
Private Data (Always Encrypted):

Data	Encryption
User credentials	AES-256-GCM
Session tokens	AES-256-GCM
Payment data	Stripe handles
RAW photographs	AES-256-GCM
Private galleries	AES-256-GCM
Contracts	AES-256-GCM
Messages	AES-256-GCM
Rule 4: RAW vs Edited Image Separation
Every uploaded photo creates TWO tracks:

RAW Track:

Original file (CR2, NEF, ARW, DNG)
Encrypted at rest
Never publicly accessible
Stored forever
Reserved for: legal proof, AI training, ownership disputes
Edited Track:

Optimized JPG/WebP
Multiple sizes (full, web, mobile, thumb)
Optional watermark
Public or private based on gallery settings

Code

├── /edited/{gallery_id}/{image_id}/

Ambient: Subtle gradients, glow pulsing, parallax

🔐 AUTHENTICATION SYSTEM {#authentication-system}
Overview
Passwordless authentication only. Users never create or manage passwords.

Authentication Methods
Method 1: Email OTP (Primary)
Flow:

Code
1. User enters email
2. System checks if email exists
   - EXISTS → Login flow
   - NOT EXISTS → Signup flow
3. Generate 6-character alphanumeric OTP
4. Store OTP hash in database with: 
   - email
   - otp_hash (bcrypt)
   - expires_at (10 minutes)
   - attempts (0)
   - ip_address
   - user_agent
5. Send email with OTP (NEVER show on screen)
6. User enters OTP from email
7. Validate: 
   - Not expired
   - Attempts < 5
   - Hash matches
8. Create/update user and session
9.  Redirect to next parameter or home
OTP Generation:

JavaScript
function generateOTP() {
  // Removed ambiguous characters:  0, O, I, 1, L
  const chars = 'ABCDEFGHJKMNPQRSTUVWXYZ23456789';
  let otp = '';
  for (let i = 0; i < 6; i++) {
    otp += chars.charAt(crypto.randomInt(chars.length));
  }
  return otp;
}

Code
Subject: Your TiredOfDoinTM verification code

Your code is: [OTP]

This code expires in 10 minutes. 

If you didn't request this, ignore this email.

Method 2: Google OAuth
Flow:

Code
1. User clicks "Sign in with Google"
2. Redirect to Google consent screen
3. User grants permission
4. Google redirects to callback with code
5. Exchange code for tokens
6. Extract user info from ID token
7. Find or create user: 
   - Search by google_id first
   - Then search by email
   - Link google_id if new to existing
   - Create new user if not found
8. Create session, redirect

Flow:

Code
1. User clicks "Forgot password" (actually "Get login code")
2. Enter email
3. Same OTP flow as login
4. On success, create new session
Session Management
Session Creation
JavaScript
async function createSession(user, request) {
  const sessionToken = crypto.randomBytes(32).toString('hex');
  const fingerprint = generateDeviceFingerprint(request);
  const expiresAt = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000); // 30 days
  
  await db.sessions.create({
    user_id: user.id,
    token_hash: hashToken(sessionToken),
    device_fingerprint: fingerprint,
    ip_address: request.ip,
    user_agent: request.headers['user-agent'],
    expires_at: expiresAt
  });
  
  response.cookie('session', sessionToken, {
    httpOnly: true,
    secure: true,
    sameSite: 'lax',
    maxAge: 30 * 24 * 60 * 60 * 1000,
    domain: 'tiredofdointm.com'
  });
  
  return session;
}
Device Fingerprinting
JavaScript
function generateDeviceFingerprint(request) {
  const components = [
    request.headers['user-agent'],
    request.headers['accept-language'],
    request.headers['accept-encoding'],
  ];
  
  return crypto
    .createHash('sha256')
    .update(components.join('|'))
    .digest('hex')
    .substring(0, 32);
}
Session Validation (Every Request)
JavaScript
async function validateSession(request) {
  const token = request.cookies.session;
  if (!token) return null;
  
  const session = await db.sessions.findOne({
    token_hash: hashToken(token)
  });
  
  if (!session) return null;
  if (new Date() > session.expires_at) {
    await db.sessions.delete(session. id);
    return null;
  }
  
  // Fingerprint validation (optional, can flag instead of block)
  const currentFingerprint = generateDeviceFingerprint(request);
  if (session.device_fingerprint !== currentFingerprint) {
    await logSecurityEvent('fingerprint_mismatch', session.user_id);
    // session.flagged = true; // or invalidate
  }
  
  const user = await db.users.findById(session.user_id);
  request.user = user;
  request.session = session;
  
  return user;
}

// Server-side validation
app.post('/api/gate/validate', (req, res) => {
  const { password } = req.body;
  
  // Constant-time comparison
  const expected = Buffer.from('67');
  const received = Buffer.from(password || '');
  
  if (expected.length !== received.length ||
      !crypto.timingSafeEqual(expected, received)) {
    return res.status(401).json({ error: 'Invalid' });
  }
  
  // Set HMAC-signed cookie
  const timestamp = Date.now();
  const expires = timestamp + (30 * 24 * 60 * 60 * 1000);
  const payload = `${timestamp}:${expires}`;
  const signature = crypto
    .createHmac('sha256', process.env.GATE_SECRET)
    .update(payload)
    .digest('hex');
  
  res.cookie('gate_pass', `${payload}:${signature}`, {
    httpOnly: true,
    secure: true,
    sameSite: 'lax',
    maxAge: 30 * 24 * 60 * 60 * 1000
  });
  
  res.json({ success: true });
});

Permission Middleware
JavaScript
function requirePermission(permission) {
  return async (req, res, next) => {
    const user = req.user;
    
    if (!user) {
      if (permission === 'view_public') return next();
      return res.redirect(`/auth? next=${encodeURIComponent(req.originalUrl)}`);
    }
    
    const role = await db.roles.findById(user.role_id);
    
    // Admin wildcard
    if (role.permissions['*']) return next();
    
    // Check specific permission
    if (role.permissions[permission]) return next();
    
    // Denied
    logSecurityEvent('permission_denied', user.id, { permission });
    return res.status(403).render('errors/403');
  };
}

🔐 GATE SYSTEM ("67")
Purpose
Simple password gate for friends/family access during development/soft launch.

User Experience Flow
Code
1. User navigates to site
       │
       ▼
2. Page begins loading normally
       │
       ▼
3. Over 1-2 seconds, screen progressively blurs
       │
       ▼
4. Sleek input panel slides up from bottom
       │
       ▼
5. User types "67"
       │
       ├── CORRECT → Blur fades, panel slides down, site accessible
       └── INCORRECT → Shake animation, red glow, try again
Security Implementation
Password validated SERVER-SIDE (cannot be bypassed via inspect element)
HMAC-signed cookie with 30-day expiration
HttpOnly + Secure + SameSite=Strict cookies
Constant-time comparison to prevent timing attacks
PHP
// Server-side validation
function validateGate(Request $request): Response {
    $password = $request->input('password');
    
    // Constant-time comparison
    $expected = getenv('GATE_PASSWORD'); // "67"
    
    if (! hash_equals($expected, $password)) {
        return response()->json(['error' => 'Invalid'], 401);
    }
    
    // Create HMAC-signed cookie
    $timestamp = time();
    $expires = $timestamp + (30 * 24 * 60 * 60); // 30 days
    $payload = "{$timestamp}:{$expires}";
    $signature = hash_hmac('sha256', $payload, getenv('GATE_SECRET'));
    
    setcookie('gate_pass', "{$payload}:{$signature}", [
        'expires' => $expires,
        'path' => '/',
        'domain' => 'tiredofdointm.com',
        'secure' => true,
        'httponly' => true,
        'samesite' => 'Strict'
    ]);
    
    return response()->json(['success' => true]);
}

OTP Generation
PHP
function generateOTP(): string {
    // Removed ambiguous characters:  0, O, I, 1, L
    $chars = 'ABCDEFGHJKMNPQRSTUVWXYZ23456789';
    $otp = '';
    for ($i = 0; $i < 6; $i++) {
        $otp .= $chars[random_int(0, strlen($chars) - 1)];
    }
    return $otp;
}

// Auto-create as admin on first deploy
async function seedAdminUsers() {
    $adminRole = Role::where('name', 'admin')->first();
    
    foreach (ADMIN_EMAILS as $email) {
        $existing = User::where('email', $email)->first();
        
        if (!$existing) {
            User::create([
                'email' => $email,
                'role_id' => $adminRole->id,
                'username' => explode('@', $email)[0],
                'email_verified' => true,
                'created_by' => 'system_seed'
            ]);
        } elseif ($existing->role_id !== $adminRole->id) {
            $existing->update(['role_id' => $adminRole->id]);
        }
    }
}
